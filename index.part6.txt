function renderRoute(){
  computeTreeRankings();buildDaySelector();
  if(!APP.rows.length){$('routeSummary').textContent='Save plantation setup first.';return}
  const today=dayTrees(APP.selectedDay),todaySet=new Set(today);
  const c=$('routeCanvas'),R=17,gap=10,rowGap=14,margin=34,maxN=Math.max(...APP.rows.map(r=>r.count));
  c.width=margin*2+maxN*(R*2+gap);c.height=margin*2+APP.rows.length*(R*2+rowGap)+45;
  const ctx=c.getContext('2d');ctx.fillStyle='#fbfdfb';ctx.fillRect(0,0,c.width,c.height);
  const pos={};
  APP.rows.forEach((row,ri)=>{
    const y=margin+ri*(R*2+rowGap)+R;
    ctx.fillStyle='#218c55';ctx.font='bold 11px system-ui';ctx.textAlign='right';ctx.fillText(row.letter,margin-7,y+4);
    for(let i=1;i<=row.count;i++)pos[row.letter+i]={x:margin+(i-1)*(R*2+gap)+R,y};
  });
  // physical row-wise route through today's selected trees
  const ordered=[];
  let forward=true;
  APP.rows.forEach(row=>{
    const arr=today.filter(k=>k[0]===row.letter).sort((a,b)=>parseInt(a.slice(1))-parseInt(b.slice(1)));
    if(!forward)arr.reverse();
    ordered.push(...arr);forward=!forward;
  });
  ctx.strokeStyle='rgba(33,140,85,.55)';ctx.lineWidth=2;
  for(let i=0;i<ordered.length-1;i++){
    const a=pos[ordered[i]],b=pos[ordered[i+1]];if(!a||!b)continue;
    ctx.beginPath();ctx.moveTo(a.x,a.y);ctx.lineTo(b.x,b.y);ctx.stroke();
  }
  APP.rows.forEach(row=>{for(let i=1;i<=row.count;i++){
    const key=row.letter+i,p=pos[key],st=APP.treeStatus[key]||'unpicked',idx=ordered.indexOf(key);
    ctx.beginPath();ctx.arc(p.x,p.y,R,0,Math.PI*2);
    if(st!=='unpicked'){ctx.fillStyle='#c3cbc4';ctx.strokeStyle='#aeb7af'}
    else if(todaySet.has(key)){ctx.fillStyle='#dff3e7';ctx.strokeStyle='#218c55'}
    else{ctx.fillStyle='#edf3eb';ctx.strokeStyle='#d0dbcf'}
    ctx.fill();ctx.lineWidth=1.5;ctx.stroke();
    ctx.fillStyle='#243026';ctx.font='bold 8px system-ui';ctx.textAlign='center';ctx.fillText(st!=='unpicked'?'✓':key,p.x,p.y+3);
    if(idx>=0){
      ctx.fillStyle='#218c55';ctx.beginPath();ctx.arc(p.x+R-3,p.y-R+3,7,0,Math.PI*2);ctx.fill();
      ctx.fillStyle='white';ctx.font='bold 7px system-ui';ctx.fillText(idx+1,p.x+R-3,p.y-R+5);
    }
  }});
  $('routeSummary').textContent=`Day ${APP.selectedDay}: ${ordered.length} trees currently unpicked · Day 1 is the next picking day; later numbers follow relative plantation ranking.`;
  c.onclick=e=>{
    if(!APP.editRoute)return;
    const rect=c.getBoundingClientRect(),x=(e.clientX-rect.left)*c.width/rect.width,y=(e.clientY-rect.top)*c.height/rect.height;
    for(const [k,p] of Object.entries(pos)){
      if(Math.hypot(x-p.x,y-p.y)<=R+3){
        APP.treeStatus[k]=(APP.treeStatus[k]||'unpicked')==='unpicked'?'picked':'unpicked';saveLocal();renderRoute();renderMap1();return;
      }
    }
  };
}

function openShareDB(){
  return new Promise((resolve,reject)=>{
    const req=indexedDB.open('dci-share-db',1);
    req.onupgradeneeded=()=>{
      const db=req.result;
      if(!db.objectStoreNames.contains('pending')) db.createObjectStore('pending',{keyPath:'id',autoIncrement:true});
    };
    req.onsuccess=()=>resolve(req.result);
    req.onerror=()=>reject(req.error);
  });
}
async function drainSharedPhotos(){
  let db;
  try{
    db=await openShareDB();
    const rows=await new Promise((resolve,reject)=>{
      const tx=db.transaction('pending','readonly');
      const req=tx.objectStore('pending').getAll();
      req.onsuccess=()=>resolve(req.result||[]);
      req.onerror=()=>reject(req.error);
    });
    if(!rows.length){db.close();return 0;}
    const files=rows.map((r,i)=>new File([r.blob],r.name||`Gallery_${Date.now()}_${i+1}.jpg`,{
      type:r.type||r.blob?.type||'image/jpeg',lastModified:r.lastModified||Date.now()
    }));
    const count=queuePickedFiles(files,'Gallery share');
    await new Promise((resolve,reject)=>{
      const tx=db.transaction('pending','readwrite');
      tx.objectStore('pending').clear();
      tx.oncomplete=resolve; tx.onerror=()=>reject(tx.error); tx.onabort=()=>reject(tx.error);
    });
    db.close();
    showSharedImportStatus(`${count} photo${count===1?'':'s'} received from Gallery and added to the queue.`,true);
    return count;
  }catch(err){
    try{db?.close()}catch(_){ }
    showSharedImportStatus('Could not read photos shared from Gallery: '+(err?.message||String(err)),false);
    return 0;
  }
}

async function setupPWA(){
  if(isStandalonePWA()) setPwaStatus('Installed · Gallery Share enabled');
  else setPwaStatus('Install once to receive Gallery shares');

  if(!('serviceWorker' in navigator)){
    setPwaStatus('Service worker unavailable in this browser'); return;
  }
  try{
    const reg=await navigator.serviceWorker.register('./sw.js',{scope:'./'});
    if(reg.waiting && navigator.serviceWorker.controller) showUpdateBanner(reg);
    reg.addEventListener('updatefound',()=>{
      const nw=reg.installing; if(!nw) return;
      nw.addEventListener('statechange',()=>{
        if(nw.state==='installed' && navigator.serviceWorker.controller) showUpdateBanner(reg);
      });
    });
    navigator.serviceWorker.addEventListener('controllerchange',()=>{
      if(pwaReloading) return; pwaReloading=true; location.reload();
    });
    try{await reg.update()}catch(_){ }
    // If launched by Android share target, the service worker stores files before redirect.
    await drainSharedPhotos();
  }catch(err){
    setPwaStatus('PWA setup failed');
    showSharedImportStatus('PWA setup error: '+(err?.message||String(err)),false);
  }
}
