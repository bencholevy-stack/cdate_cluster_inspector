const TRAIN_DB='dci_training_v1',TRAIN_STORE='examples';
function openTrainingDB(){
  return new Promise((resolve,reject)=>{
    if(!('indexedDB' in window)){reject(new Error('IndexedDB unavailable'));return;}
    const req=indexedDB.open(TRAIN_DB,1);
    req.onupgradeneeded=()=>{const db=req.result;if(!db.objectStoreNames.contains(TRAIN_STORE))db.createObjectStore(TRAIN_STORE,{keyPath:'id'});};
    req.onsuccess=()=>resolve(req.result);req.onerror=()=>reject(req.error||new Error('Training database error'));
  });
}

}
