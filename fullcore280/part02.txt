function compactResultsForStorage(){
  const out={};
  for(const [key,t] of Object.entries(APP.results)){
    const sides={};
    for(const [side,r] of Object.entries(t.sides||{})){
      sides[side]={
        name:r.name||'',tree:r.tree||t.tree||null,side:r.side||side,
        pct:r.pct||null,confidence:r.confidence||0
      };
    }
    out[key]={tree:t.tree||null,sides,avg:t.avg||null,rank:t.rank||null,dayBucket:t.dayBucket||null};
  }
  return out;
}

const TRAIN_DB='dci_training_v1',TRAIN_STORE='examples';
function openTrainingDB(){
  return new Promise((resolve,reject)=>{
    if(!('indexedDB' in window)){reject(new Error('IndexedDB unavailable'));return;}
    const req=indexedDB.open(TRAIN_DB,1);
    req.onupgradeneeded=()=>{const db=req.result;if(!db.objectStoreNames.contains(TRAIN_STORE))db.createObjectStore(TRAIN_STORE,{keyPath:'id'});};
    req.onsuccess=()=>resolve(req.result);req.onerror=()=>reject(req.error||new Error('Training database error'));
  });
}
async function dbAllTraining(){
  const db=await openTrainingDB();return new Promise((resolve,reject)=>{const tx=db.transaction(TRAIN_STORE,'readonly'),req=tx.objectStore(TRAIN_STORE).getAll();req.onsuccess=()=>resolve(req.result||[]);req.onerror=()=>reject(req.error);});
}
async function dbPutTraining(ex){
  const db=await openTrainingDB();return new Promise((resolve,reject)=>{const tx=db.transaction(TRAIN_STORE,'readwrite');tx.objectStore(TRAIN_STORE).put(ex);tx.oncomplete=()=>resolve();tx.onerror=()=>reject(tx.error);});
}
async function dbClearTraining(){
  const db=await openTrainingDB();return new Promise((resolve,reject)=>{const tx=db.transaction(TRAIN_STORE,'readwrite');tx.objectStore(TRAIN_STORE).clear();tx.oncomplete=()=>resolve();tx.onerror=()=>reject(tx.error);});
}

}

  const v=$('view-'+tab);if(v)v.classList.add('on');
  if(tab==='map1') renderMap1();
  if(tab==='route') renderRoute();
  if(tab==='analyze') updatePlantationPill();
  window.scrollTo({top:0,behavior:'smooth'});
}
function switchTab(tab,fromHistory=false){
  if(!fromHistory && tab!==APP.currentTab){
    try{history.pushState({dci:true,kind:'tab',tab},'',location.href)}catch(e){}
  }
  activateTab(tab);
}
document.querySelectorAll('.tab').forEach(b=>b.addEventListener('click',()=>switchTab(b.dataset.tab)));
function initBackGuard(){
  if(NAV.initialized)return;NAV.initialized=true;
  try{
    history.replaceState({dci:true,kind:'base',tab:APP.currentTab},'',location.href);
    history.pushState({dci:true,kind:'guard',tab:APP.currentTab},'',location.href);
  }catch(e){}
}
window.addEventListener('popstate',function(e){
  if($('viewerModal').classList.contains('on')){$('viewerModal').classList.remove('on');return;}
  if($('treeModal').classList.contains('on')){closeTreeDetails(true);return;}
  const st=e.state||{};
  if((st.kind==='tab'||st.kind==='guard')&&st.tab){activateTab(st.tab);return;}
  if(st.kind==='base'){
    activateTab(st.tab||'setup');
    try{history.pushState({dci:true,kind:'guard',tab:APP.currentTab},'',location.href)}catch(err){}
    toast('Back stays inside Date Cluster Inspector');
    return;
  }
  try{history.pushState({dci:true,kind:'guard',tab:APP.currentTab},'',location.href)}catch(err){}
});

function buildRowsTable(){
  const n=Math.max(1,Math.min(26,parseInt($('nrows').value)||1));
  const def=Math.max(1,parseInt($('defaultTrees').value)||20);
  const old={}; APP.rows.forEach(r=>old[r.letter]=r.count);
  $('rowsBox').innerHTML='';
  for(let i=0;i<n;i++){
    const l=String.fromCharCode(65+i);
    const div=document.createElement('div'); div.className='rowline';
    div.innerHTML=`<div class="rowletter">Row ${l}</div><input id="row-${l}" type="number" min="1" value="${old[l]||def}">`;
    $('rowsBox').appendChild(div);
  }
}
$('buildRowsBtn').onclick=buildRowsTable;
$('saveSetupBtn').onclick=()=>{
  const n=parseInt($('nrows').value)||0;
  const rows=[];
  for(let i=0;i<n;i++){
    const l=String.fromCharCode(65+i), el=$('row-'+l), count=parseInt(el?.value)||0;
    if(count<1){toast('Fill all tree counts');return}
    rows.push({letter:l,count});
  }
  APP.rows=rows;
  APP.treesPerDay=Math.max(1,parseInt($('treesPerDay').value)||10);
  APP.rows.forEach(r=>{for(let i=1;i<=r.count;i++){const k=r.letter+i;if(!APP.treeStatus[k])APP.treeStatus[k]='unpicked'}});
  saveLocal(); computeTreeRankings(); updatePlantationPill(); toast('Plantation saved');
};
$('resetPickingBtn').onclick=()=>{

  APP.treeStatus={}; saveLocal(); renderMap1(); renderRoute(); toast('Picking status reset');
};
