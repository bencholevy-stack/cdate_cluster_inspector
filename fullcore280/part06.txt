=w;t.height=h;
  const ctx=t.getContext('2d',{willReadFrequently:true});ctx.drawImage(c,0,0,w,h);
  const d=ctx.getImageData(0,0,w,h).data;
  let sum=0,n=0;
  for(let y=1;y<h-1;y+=2)for(let x=1;x<w-1;x+=2){
    const i=(y*w+x)*4;
    const g0=(d[i]+d[i+1]+d[i+2])/3;
    const ids=[((y-1)*w+x)*4,((y+1)*w+x)*4,(y*w+x-1)*4,(y*w+x+1)*4];
    let lap=-4*g0;
    ids.forEach(j=>lap+=(d[j]+d[j+1]+d[j+2])/3);
    sum+=lap*lap;n++;
  }
  return n?sum/n:0;
}
async function imageFileToFrame(file){
  return new Promise((resolve,reject)=>{
    const url=URL.createObjectURL(file),img=new Image();
    img.onload=()=>{
      const c=document.createElement('canvas');c.width=img.naturalWidth;c.height=img.naturalHeight;
      c.getContext('2d').drawImage(img,0,0);
      const raw=c.toDataURL('image/jpeg',.92);
      URL.revokeObjectURL(url);
      resolve([{canvas:c,raw,sharpness:sharpnessOfCanvas(c),time:0}]);
    };
    img.onerror=()=>{URL.revokeObjectURL(url);reject(new Error('Cannot read image'))};
    img.src=url;
  });
}
async function videoFileToFrames(file){
  return new Promise((resolve,reject)=>{
    const url=URL.createObjectURL(file),v=document.createElement('video');
    v.muted=true;v.playsInline=true;v.preload='metadata';
    const timeout=setTimeout(()=>{URL.revokeObjectURL(url);reject(new Error('Video timed out'))},45000);
    v.onloadedmetadata=()=>{
      const dur=Math.max(.5,v.duration||4);
      const sampleN=Math.max(6,APP.settings.framesPerVideo*3);
      const times=Array.from({length:sampleN},(_,i)=>dur*(.08+(i/(sampleN-1))*.84));
      const out=[];let idx=0;
      const capture=()=>{
        if(idx>=times.length){
          clearTimeout(timeout);URL.revokeObjectURL(url);
          out.sort((a,b)=>b.sharpness-a.sharpness);
          resolve(out.slice(0,APP.settings.framesPerVideo).sort((a,b)=>a.time-b.time));
          return;
        }
        v.currentTime=Math.min(dur-.02,times[idx]);
      };
      v.onseeked=()=>{
        try{
          const maxDim=1400,scale=Math.min(1,maxDim/Math.max(v.videoWidth||640,v.videoHeight||480));
          const c=document.createElement('canvas');c.width=Math.round((v.videoWidth||640)*scale);c.height=Math.round((v.videoHeight||480)*scale);
          c.getContext('2d').drawImage(v,0,0,c.width,c.height);
          out.push({canvas:c,raw:c.toDataURL('image/jpeg',.9),sharpness:sharpnessOfCanvas(c),time:times[idx]});
        }catch(e){}
        idx++;capture();
      };
      capture();
    };
    v.onerror=()=>{clearTimeout(timeout);URL.revokeObjectURL(url);reject(new Error('Cannot read video'))};
    v.src=url;
  });
}

function aggregateFrames(frames){
  const usable=frames.filter(f=>f.analysisValid!==false);
  if(!usable.length)return {pct:{green:0,yellow:0,orange:0,brown:0,dry:0,unknown:100},fruitEq:{green:0,yellow:0,brown:0,dry:0,unknown:0,totalValid:0,totalDetected:0,typicalShort:0,typicalArea:0},roiCoverage:0,confidence:0,analysisValid:false,invalidReason:'No reliable cluster ROI in this file.'};
  frames=usable;
  const keys=['green','yellow','orange','brown','dry'];
  const weights=frames.map(f=>Math.max(.05,f.confidence));
  const sw=weights.reduce((a,b)=>a+b,0);
  const pct={};
  keys.forEach(k=>pct[k]=frames.reduce((s,f,i)=>s+f.pct[k]*weights[i],0)/sw);
  pct.unknown=frames.reduce((s,f,i)=>s+f.pct.unknown*weights[i],0)/sw;
  const fruitEq={};
  for(const k of ['green','yellow','brown','dry','unknown','totalValid','totalDetected','typicalShort','typicalArea'])fruitEq[k]=frames.reduce((s,f,i)=>s+((f.fruitEq?.[k]||0)*weights[i]),0)/sw;
  const roiCoverage=frames.reduce((s,f,i)=>s+((f.roi?.coverage||0)*weights[i]),0)/sw;
  return {pct,fruitEq,roiCoverage,confidence:frames.reduce((s,f,i)=>s+f.confidence*weights[i],0)/sw,analysisValid:true};
}
function combineTreeSides(){
  Object.values(APP.results).forEach(t=>{
    const sides=Object.values(t.sides||{}).filter(r=>r.analysisValid!==false);
    if(!sides.length){t.avg=null;return;}
    const keys=['green','yellow','orange','brown','dry','unknown'];
    const avg={pct:{},confidence:sides.reduce((s,r)=>s+r.confidence,0)/sides.length,fruitEq:{}};
    keys.forEach(k=>avg.pct[k]=sides.reduce((s,r)=>s+r.pct[k],0)/sides.length);

    avg.ripenessScore=avg.pct.brown;
    t.avg=avg;
  });
}

function rankingMetrics(t){
  const p=t?.avg?.pct||{green:0,yellow:0,orange:0,brown:0,unknown:100};
  const gyDen=p.green+p.yellow;
  const ybDen=p.yellow+p.brown;
  const yellowProgress=gyDen>0?100*p.yellow/gyDen:0;
  const brownProgress=ybDen>0?100*p.brown/ybDen:0;
  const brownPresent=p.brown>=(APP.settings.brownPresenceMin||1.5);
  let raw=0,basis='';
  const mode=APP.settings.rankMode||'auto';
  if(mode==='brown'){
    raw=brownProgress;basis='Brown stage: more brown = more urgent';
  }else if(mode==='yellow'){
    if(brownPresent){
      const yellowFrac=ybDen>0?100*p.yellow/ybDen:100;
      raw=100-yellowFrac;basis='Yellow + brown: more yellow = less urgent';
    }else{
      raw=yellowProgress;basis='Green + yellow: more yellow = more urgent';
    }
  }else{
    if(brownPresent){raw=brownProgress;basis='Auto → yellow-to-brown stage';}
    else{raw=yellowProgress;basis='Auto → green-to-yellow stage';}
  }
  return {rawUrgency:Math.max(0,Math.min(100,raw)),basis,yellowProgress,brownProgress,brownPresent};
}
function computeTreeRankings(){
  combineTreeSides();
  const trees=Object.values(APP.results).filter(t=>t.tree&&t.avg&&t.avg.pct);
  trees.forEach(t=>Object.assign(t,rankingMetrics(t)));
  trees.sort((a,b)=>b.rawUrgency-a.rawUrgency || (b.avg.confidence||0)-(a.avg.confidence||0) || String(a.tree).localeCompare(String(b.tree),undefined,{numeric:true}));
  const N=trees.length;
  const avgRaw=N?trees.reduce((s,t)=>s+t.rawUrgency,0)/N:0;
  const avgYellow=N?trees.reduce((s,t)=>s+(t.avg.pct.yellow||0),0)/N:0;
  const avgBrown=N?trees.reduce((s,t)=>s+(t.avg.pct.brown||0),0)/N:0;
  trees.forEach((t,i)=>{
    t.rank=i+1;
    // Relative plantation percentage: top tree = 100%; with 20 trees the last = 5%.
    const below=trees.filter(x=>x.rawUrgency<t.rawUrgency-1e-9).length;
    t.relativePct=Math.round(100*(below+1)/Math.max(1,N));
    t.dayBucket=Math.floor(i/Math.max(1,APP.treesPerDay))+1;
    t.plantationAvgRaw=avgRaw;t.plantationAvgYellow=avgYellow;t.plantationAvgBrown=avgBrown;t.totalAnalyzed=N;
  });
}
