const TEACH_COLORS={green:'#22a65b',yellow:'#e1b400',brown:'#8a5833',dry:'#78668f',unknown:'#8b9790',notfruit:'#d44c4c'};
function renderTeachImage(){
  if(!APP.viewer.teaching||!APP.viewer.rawSrc){$('viewerImg').src=APP.viewer.src;return;}
  const base=new Image();
  base.onload=()=>{
    const c=document.createElement('canvas');c.width=base.naturalWidth;c.height=base.naturalHeight;const ctx=c.getContext('2d');ctx.drawImage(base,0,0);
    const scale=Math.max(1,Math.min(c.width,c.height)/700),rad=9*scale;
    (APP.viewer.corrections||[]).forEach((l,i)=>{
      const x=l.x*c.width,y=l.y*c.height,col=TEACH_COLORS[l.cls]||'#fff';
      ctx.save();ctx.lineWidth=3*scale;ctx.strokeStyle=col;ctx.fillStyle='rgba(255,255,255,.88)';
      if(l.cls==='notfruit'){
        ctx.beginPath();ctx.moveTo(x-rad,y-rad);ctx.lineTo(x+rad,y+rad);ctx.moveTo(x+rad,y-rad);ctx.lineTo(x-rad,y+rad);ctx.stroke();
      }else{
        ctx.beginPath();ctx.arc(x,y,rad,0,Math.PI*2);ctx.fill();ctx.stroke();
        ctx.fillStyle=col;ctx.font=`bold ${10*scale}px sans-serif`;ctx.textAlign='center';ctx.textBaseline='middle';
        const letter={green:'G',yellow:'Y',brown:'B',dry:'D',unknown:'?'}[l.cls]||'?';ctx.fillText(letter,x,y);
        if(l.partial){ctx.setLineDash([4*scale,3*scale]);ctx.beginPath();ctx.arc(x,y,rad*1.55,0,Math.PI*2);ctx.stroke();}
      }
      ctx.restore();
    });
    $('viewerImg').src=c.toDataURL('image/jpeg',.92);
  };base.src=APP.viewer.rawSrc;
}

function manualFrameMetrics(labels,oldFrame){
  const c={green:0,yellow:0,brown:0,dry:0,unknown:0};labels.forEach(l=>{if(c[l.cls]!==undefined)c[l.cls]++;});
  const valid=c.green+c.yellow+c.brown,totalDetected=valid+c.dry+c.unknown,pct={green:0,yellow:0,orange:0,brown:0,dry:0,unknown:0};
  if(valid){pct.green=100*c.green/valid;pct.yellow=100*c.yellow/valid;pct.brown=100*c.brown/valid;}
  pct.dry=totalDetected?100*c.dry/totalDetected:0;pct.unknown=(valid+c.unknown)?100*c.unknown/(valid+c.unknown):0;
  return {pct,fruitEq:{...c,totalValid:valid,totalDetected,typicalShort:oldFrame?.fruitEq?.typicalShort||0,typicalArea:oldFrame?.fruitEq?.typicalArea||0},confidence:Math.max(0,1-pct.unknown/100),analysisValid:valid>0,invalidReason:valid>0?'':'Fully corrected frame contains no valid fruit labels.'};
}

function heatColor(p){
  p=Math.max(0,Math.min(100,p||0));
  const stops=[[0,[220,239,220]],[45,[244,223,115]],[72,[239,154,72]],[100,[217,91,63]]];
  for(let i=0;i<stops.length-1;i++){
    const a=stops[i],b=stops[i+1];if(p>=a[0]&&p<=b[0]){
      const t=(p-a[0])/(b[0]-a[0]);const c=a[1].map((v,j)=>Math.round(v+(b[1][j]-v)*t));return `rgb(${c.join(',')})`;
    }
  }return 'rgb(217,91,63)';
}
